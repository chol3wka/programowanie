Procedura Scalaj(Poczatek,Koniec)
Dla (i=Poczatek,...,Koniec) Wykonaj
  {
  DanePom[i] <- Dane[i]
  }
p <- Poczatek
q <- (Poczatek+Koniec) / 2 + 1
r <- Poczatek
Dopóki (p<=((Poczatek+Koniec) / 2) and (q <= Koniec)) Wykonaj
 {
 Je¿eli (DanePom[r] < DanePom[q])
    {
    DanePom[r] < DanePom[p]
    r <- r + 1
    p <- p + 1
    }
 W przeciwnym razie
    {
    DanePom[r] < DanePom[q]
    r <- r + 1
    q <- q + 1    
    }
 }
Dopóki (p<=(Poczatek+Koniec)/2) Wykonaj
 {
 Dane[r] < DanePom[p]
 r <- r + 1
 p <- p + 1    
 }


Procedura MergeSort(Poczatek,Koniec)
Je¿eli (Poczatek<Koniec)

  MergeSort(Poczatek,(Poczatek+Koniec)/2)
  MergeSort((Poczatek+Koniec)/2+1,Koniec)
  Scalaj(Poczatek,Koniec)
  }
