proc Scalaj(Poczatek,Koniec)
for (i=Poczatek,...,Koniec) do
  {
  DanePom[i] <- Dane[i]
  }
p <- Poczatek
q <- (Poczatek+Koniec) / 2 + 1
r <- Poczatek
while (p<=((Poczatek+Koniec) / 2) and (q <= Koniec)) do
 {
 if (DanePom[r] < DanePom[q])
    {
    DanePom[r] < DanePom[p]
    r <- r + 1
    p <- p + 1
    }
 else
    {
    DanePom[r] < DanePom[q]
    r <- r + 1
    q <- q + 1    
    }
 }
while (p<=(Poczatek+Koniec)/2) do
 {
 Dane[r] < DanePom[p]
 r <- r + 1
 p <- p + 1    
 }

  MergeSort((Poczatek+Koniec)/2+1,Koniec)
  Scalaj(Poczatek,Koniec)
