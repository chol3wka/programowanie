proc Sterta(array, size, index)
largestIndex <- index
leftChild <- 2 * index + 1
rightChild <- 2 * index + 2
if (leftChild < size and array[leftChild] > array[largestIndex])
    {
    largestIndex <- leftChild
    }
if (rightChild < size and array[rightChild] > array[largestIndex])
    {
    largestIndex <- rightChild
    }
if (largestIndex <> index)
    {
    tempVar <- array[index]
    array[index] <- array[largestIndex]
    array[largestIndex] <- tempVar
    Sterta(array, size, largestIndex)
    }

fun HeapSort(array, size)
if (size <= 1)
   return array
for (i=size / 2 - 1,...,0) do
    Sterta(array, size, i)
for (i=size-1,...,0) do
    {
    tempVar <- array[0]
    array[0] <- array[i]
    array[i] <- tempVar
    Sterta(array, i, 0)
    }
return array
