Procedura Sterta(array, size, index)
largestIndex <-- index
leftChild <-- 2 * index + 1
rightChild <-- 2 * index + 2
Je¿eli (leftChild < size and array[leftChild] > array[largestIndex])
    {
    largestIndex <-- leftChild
    }
Je¿eli (rightChild < size and array[rightChild] > array[largestIndex])
    {
    largestIndex <-- rightChild
    }
Je¿eli (largestIndex <> index)
    {
    tempVar <-- array[index]
    array[index] <-- array[largestIndex]
    array[largestIndex] <-- tempVar
    Sterta(array, size, largestIndex)
    }
